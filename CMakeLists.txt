cmake_minimum_required(VERSION 3.16)
project(recomp3ds CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(UNICORN_ROOT "$ENV{UNICORN_ROOT}" CACHE PATH "unicorn root (include/ + lib/), e.g. <venv>/lib/python3.x/site-packages/unicorn")
file(GLOB RT_SRC rt/*.cpp)
file(GLOB TRANSLATOR_SRC web/translator/*.js)
if(EMSCRIPTEN)
  # every object (lifted code included) must be built with threads enabled
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -pthread")
  set(R3DS_UNICORN OFF CACHE BOOL "" FORCE)
endif()
option(R3DS_LIFTED "link lifted code from gen/" ON)
if(R3DS_LIFTED AND EXISTS "${CMAKE_SOURCE_DIR}/gen/CMakeLists.txt")
  add_subdirectory(gen)
endif()
option(R3DS_UNICORN "use Unicorn instead of the built-in interpreter as the fallback" OFF)
add_executable(recomp3ds ${RT_SRC})
target_include_directories(recomp3ds PRIVATE rt)
# character tables of the game's fonts (tools/gen_fonts.py, from the ROM): high-resolution text
if(EXISTS "${CMAKE_SOURCE_DIR}/gen/fonts_gen.cpp")
  target_sources(recomp3ds PRIVATE gen/fonts_gen.cpp)
  target_compile_definitions(recomp3ds PRIVATE R3DS_HAVE_FONTS=1)
endif()
if(APPLE AND NOT EMSCRIPTEN)
  target_link_libraries(recomp3ds PRIVATE "-framework CoreText" "-framework CoreGraphics" "-framework CoreFoundation")
endif()
if(EMSCRIPTEN)
  target_compile_options(recomp3ds PRIVATE -msimd128 -msse2 --use-port=emdawnwebgpu)
  target_compile_definitions(recomp3ds PRIVATE R3DS_WEBGPU=1)
  target_link_options(recomp3ds PRIVATE --use-port=emdawnwebgpu)
  # the ARM -> WASM translator (web/translator) runs inside the runtime's own threads
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/translator.bundle.js
    COMMAND python3 ${CMAKE_SOURCE_DIR}/tools/bundle_translator.py -o ${CMAKE_BINARY_DIR}/translator.bundle.js
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/bundle_translator.py ${TRANSLATOR_SRC})
  add_custom_target(translator_bundle DEPENDS ${CMAKE_BINARY_DIR}/translator.bundle.js)
  add_dependencies(recomp3ds translator_bundle)
  target_link_options(recomp3ds PRIVATE --pre-js ${CMAKE_BINARY_DIR}/translator.bundle.js -sALLOW_TABLE_GROWTH=1)
  set_property(TARGET recomp3ds APPEND PROPERTY LINK_DEPENDS ${CMAKE_BINARY_DIR}/translator.bundle.js)
  target_link_options(recomp3ds PRIVATE
    -pthread -sPROXY_TO_PTHREAD=1 -sPTHREAD_POOL_SIZE=24 -sPTHREAD_POOL_SIZE_STRICT=0
    -sINITIAL_MEMORY=1073741824 -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=4294967296
    -sSTACK_SIZE=4194304 -sDEFAULT_PTHREAD_STACK_SIZE=2097152
    -sMODULARIZE=1 -sEXPORT_NAME=createR3DS -sENVIRONMENT=web,worker
    -sEXIT_RUNTIME=0 -sALLOW_BLOCKING_ON_MAIN_THREAD=1
    "-sEXPORTED_RUNTIME_METHODS=['FS','ENV','IDBFS','wasmMemory','HEAPU8','addRunDependency','removeRunDependency']"
    "-sEXPORTED_FUNCTIONS=['_main','_malloc','_free','_gh_step','_gh_svc','_web_rom_ctl','_web_frame','_web_frame_count','_web_input','_web_audio_ring','_web_bottom_coverage','_web_bottom_masked','_web_hwr_start','_web_hwr_frame','_web_hwr_state','_web_hwr_out','_web_hwr_soft_alpha','_web_debug_stats']"
    -lidbfs.js)
  option(R3DS_WEB_DEBUG "web build with Emscripten assertions" OFF)
  if(R3DS_WEB_DEBUG)
    target_link_options(recomp3ds PRIVATE -sASSERTIONS=2 -g2)
  endif()
  set_target_properties(recomp3ds PROPERTIES SUFFIX ".js")
  file(GLOB WEB_FILES ${CMAKE_SOURCE_DIR}/web/*)
  add_custom_command(TARGET recomp3ds POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy ${WEB_FILES} $<TARGET_FILE_DIR:recomp3ds>)
elseif(WIN32)
  target_link_options(recomp3ds PRIVATE -static)
else()
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)$")
    # ARM64 hosts (Apple Silicon, Linux arm64): the shader interpreter's SSE2
    # intrinsics go through sse2neon. No FMA contraction, as on x86-64.
    find_path(SSE2NEON_INCLUDE_DIR sse2neon.h HINTS /opt/homebrew/include /usr/local/include)
    if(NOT SSE2NEON_INCLUDE_DIR)
      message(FATAL_ERROR "ARM64 build needs sse2neon.h: brew install sse2neon (macOS) or apt install sse2neon")
    endif()
    target_include_directories(recomp3ds PRIVATE ${SSE2NEON_INCLUDE_DIR})
    target_compile_options(recomp3ds PRIVATE -ffp-contract=off)
  endif()
  find_package(Threads REQUIRED)
  target_link_libraries(recomp3ds PRIVATE Threads::Threads)
endif()
if(R3DS_UNICORN)
  target_compile_definitions(recomp3ds PRIVATE R3DS_USE_UNICORN=1)
  target_include_directories(recomp3ds PRIVATE ${UNICORN_ROOT}/include)
  if(WIN32)
    target_link_libraries(recomp3ds PRIVATE ${UNICORN_ROOT}/lib/unicorn.dll)
  else()
    target_link_libraries(recomp3ds PRIVATE ${UNICORN_ROOT}/lib/libunicorn.so.2)
    set_target_properties(recomp3ds PROPERTIES BUILD_RPATH "${UNICORN_ROOT}/lib")
  endif()
endif()
if(NOT EMSCRIPTEN)
  find_package(SDL2 QUIET)
  # WebGPU renderer (wgpu-native): brew install wgpu-native, or point WGPU_NATIVE_ROOT at a release
  option(R3DS_WEBGPU "GPU renderer through wgpu-native when available" ON)
  if(R3DS_WEBGPU)
    find_path(WGPU_INCLUDE_DIR webgpu.h HINTS ${WGPU_NATIVE_ROOT}/include /opt/homebrew/opt/wgpu-native/include /usr/local/opt/wgpu-native/include PATH_SUFFIXES webgpu)
    find_library(WGPU_LIBRARY wgpu_native HINTS ${WGPU_NATIVE_ROOT}/lib /opt/homebrew/opt/wgpu-native/lib /usr/local/opt/wgpu-native/lib)
    if(WGPU_INCLUDE_DIR AND WGPU_LIBRARY)
      target_include_directories(recomp3ds PRIVATE ${WGPU_INCLUDE_DIR})
      target_link_libraries(recomp3ds PRIVATE ${WGPU_LIBRARY})
      target_compile_definitions(recomp3ds PRIVATE R3DS_WEBGPU=1)
      get_filename_component(WGPU_LIBDIR ${WGPU_LIBRARY} DIRECTORY)
      set_target_properties(recomp3ds PROPERTIES BUILD_RPATH "${WGPU_LIBDIR}")
      message(STATUS "WebGPU renderer enabled (${WGPU_LIBRARY})")
    else()
      message(STATUS "wgpu-native not found: software renderer only")
    endif()
  endif()
endif()
if(SDL2_FOUND)
  if(TARGET SDL2::SDL2)
    target_link_libraries(recomp3ds PRIVATE SDL2::SDL2)   # not SDL2main: we define SDL_MAIN_HANDLED
  else()
    target_include_directories(recomp3ds PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(recomp3ds PRIVATE ${SDL2_LIBRARIES})
  endif()
  target_compile_definitions(recomp3ds PRIVATE R3DS_HAVE_SDL=1)
  message(STATUS "SDL2 frontend enabled")
endif()
if(TARGET r3dslifted)
  target_link_libraries(recomp3ds PRIVATE r3dslifted)
  target_compile_definitions(recomp3ds PRIVATE R3DS_HAVE_LIFTED=1)
endif()
